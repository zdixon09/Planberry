    <script>
        // --- GLOBAL APP STATE & PERSISTENCE ---
        let appState = {
            pin: "4125",
            theme: "strawberry",
            currentYear: 2026,
            students: [
                { id: 1, name: "Zoe Smith", grade: 85, scale: "P", iep: "Visual Schedule" },
                { id: 2, name: "Alex Jones", grade: 62, scale: "D", iep: "Extra Time" }
            ],
            anecdotals: {},
            selectedStandards: []
        };

        const BC_CURRICULUM = {
            "3": {
                "MATH": ["Demonstrate fluency with addition facts to 20", "Understand fraction concepts as part of a whole"],
                "ELA": ["Use foundational reading strategies to comprehend text", "Create stories with clear structure"]
            }
        };

        let targetViewAfterPin = "";

        // --- INITIALIZATION ---
        window.addEventListener('DOMContentLoaded', () => {
            loadState();
            initDateDisplay();
            renderMiniCalendar();
            renderYearCalendar(appState.currentYear);
            renderStudentTable();
            initGrowthChart();
            setupAutosave();
        });

        function saveState() {
            localStorage.setItem('kawaii_planner_data', JSON.stringify(appState));
        }

        function loadState() {
            const saved = localStorage.getItem('kawaii_planner_data');
            if(saved) {
                appState = Object.assign(appState, JSON.parse(saved));
            }
            setTheme(appState.theme);
        }

        function setupAutosave() {
            document.querySelectorAll('[contenteditable="true"]').forEach(el => {
                el.addEventListener('input', () => saveState());
            });
        }

        // --- UI & NAVIGATION ---
        function switchView(viewId) {
            document.querySelectorAll('.view-panel').forEach(el => el.classList.remove('active'));
            document.querySelectorAll('.nav-btn').forEach(el => el.classList.remove('active'));
            
            const target = document.getElementById(`view-${viewId}`);
            if(target) target.classList.add('active');
            
            saveState();
        }

        function setTheme(themeName) {
            document.documentElement.setAttribute('data-theme', themeName);
            appState.theme = themeName;
            saveState();
        }

        function initDateDisplay() {
            const options = { weekday: 'long', year: 'numeric', month: 'short', day: 'numeric' };
            document.getElementById('current-date-badge').innerText = new Date().toLocaleDateString('en-US', options);
        }

        // --- SECURITY PIN HANDLER ---
        function promptPinForView(viewId) {
            targetViewAfterPin = viewId;
            document.getElementById('pin-input').value = "";
            document.getElementById('pin-modal').style.display = 'flex';
        }

        function verifyPin() {
            const entered = document.getElementById('pin-input').value;
            if(entered === appState.pin) {
                closeModal('pin-modal');
                switchView(targetViewAfterPin);
            } else {
                alert("Incorrect PIN! Default is 4125");
            }
        }

        function closeModal(id) {
            document.getElementById(id).style.display = 'none';
        }

        // --- CALENDAR & TIMETABLE SYSTEM ---
        function renderMiniCalendar() {
            const container = document.getElementById('mini-calendar-container');
            container.innerHTML = "<p style='font-size:0.85rem; color:var(--text-muted);'>Active Week: Oct 12 - Oct 16, 2026</p>";
        }

        function renderYearCalendar(year) {
            document.getElementById('calendar-year-display').innerText = year;
            const grid = document.getElementById('year-grid');
            grid.innerHTML = '';
            
            const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
            months.forEach((m, idx) => {
                const card = document.createElement('div');
                card.style.cssText = "border:1px solid var(--border-color); padding:10px; border-radius:12px; text-align:center; cursor:pointer;";
                card.innerHTML = `<strong>${m}</strong><br><span style="font-size:0.75rem;">Interactive View</span>`;
                card.onclick = () => { switchView('daily'); document.getElementById('daily-date-title').innerText = `${m} Daily Planning`; };
                grid.appendChild(card);
            });
        }

        function changeYear(delta) {
            appState.currentYear += delta;
            renderYearCalendar(appState.currentYear);
        }

        // --- STANDARDS MODAL & BC CURRICULUM ---
        function openStandardsModal() {
            loadCurriculumOptions();
            document.getElementById('standards-modal').style.display = 'flex';
        }

        function loadCurriculumOptions() {
            const g = document.getElementById('std-grade').value;
            const s = document.getElementById('std-subject').value;
            const container = document.getElementById('standards-checklist');
            container.innerHTML = '';

            const list = (BC_CURRICULUM[g] && BC_CURRICULUM[g][s]) ? BC_CURRICULUM[g][s] : ["Standard outcome placeholder standard A", "Standard outcome placeholder standard B"];

                container.innerHTML += `<label><input type="checkbox" value="BC-${s}-${g}-${idx}"> ${item}</label><br>`;
            });

            const container = document.getElementById('connected-standards-badges');
            container.innerHTML = '';

                container.innerHTML += `<span class="pill-tag">${cb.value}</span>`;
            });

        }

        // --- @STUDENT SMART ANECDOTAL ENGINE ---
        function handleMentionSearch(e) {
            const sel = window.getSelection();
            if(!sel.rangeCount) return;
            const text = sel.anchorNode.textContent;
            
            if(text.includes('@')) {
                const dropdown = document.getElementById('mention-dropdown');
                dropdown.style.display = 'block';
                dropdown.style.top = (e.clientY + 15) + 'px';
                dropdown.style.left = e.clientX + 'px';
                
                dropdown.innerHTML = '';
                appState.students.forEach(st => {
                    const item = document.createElement('div');
                    item.className = 'mention-item';
                    item.innerText = `@${st.name}`;
                    item.onclick = () => {
                        document.execCommand('insertText', false, st.name + ' - ');
                        dropdown.style.display = 'none';
                    };
                    dropdown.appendChild(item);
                });
            } else {
                document.getElementById('mention-dropdown').style.display = 'none';
            }
        }

        // --- STUDENT HUB & GRADEBOOK ---
        function renderStudentTable() {
            const tbody = document.getElementById('student-table-body');
            tbody.innerHTML = '';
            
            appState.students.forEach(st => {
                let badgeColor = st.grade >= 80 ? '#C1E1C1' : st.grade >= 60 ? '#FFFACD' : '#FFD1DC';
                tbody.innerHTML += `
                    <tr>
                        <td><strong>${st.name}</strong></td>
                        <td><input type="number" value="${st.grade}" style="width:60px;" onchange="updateStudentGrade(${st.id}, this.value)">%</td>
                        <td><span class="pill-tag" style="background:${badgeColor}">${st.scale}</span></td>
                        <td><input type="text" value="${st.iep}" onchange="updateStudentIEP(${st.id}, this.value)"></td>
                        <td><button class="btn btn-secondary" style="font-size:0.75rem;" onclick="alert('Viewing anecdotal entries for ${st.name}')">View Logs</button></td>
                    </tr>
                `;
            });
        }

        function updateStudentGrade(id, val) {
            const st = appState.students.find(s => s.id === id);
            if(st) {
                st.grade = parseInt(val) || 0;
                st.scale = st.grade >= 80 ? 'EX' : st.grade >= 70 ? 'P' : st.grade >= 50 ? 'D' : 'EM';
                renderStudentTable();
                saveState();
            }
        }

        function updateStudentIEP(id, val) {
            const st = appState.students.find(s => s.id === id);
            if(st) { st.iep = val; saveState(); }
        }

                renderStudentTable();
                saveState();
            }

        // --- STICKERS ENGINE ---
        function spawnSticker(emoji) {
            const sticker = document.createElement('div');
            sticker.className = 'placed-sticker';
            sticker.innerText = emoji;
            sticker.style.left = '300px';
            sticker.style.top = '200px';
            
            // Drag logic
            let isDragging = false;
            sticker.addEventListener('mousedown', (e) => {
                isDragging = true;
                let shiftX = e.clientX - sticker.getBoundingClientRect().left;
                let shiftY = e.clientY - sticker.getBoundingClientRect().top;
                
                function moveAt(pageX, pageY) {
                    sticker.style.left = pageX - shiftX + 'px';
                    sticker.style.top = pageY - shiftY + 'px';
                }
                
                function onMouseMove(event) {
                    if (isDragging) moveAt(event.pageX, event.pageY);
                }
                
                document.addEventListener('mousemove', onMouseMove);
                
                sticker.onmouseup = function() {
                    isDragging = false;
                    document.removeEventListener('mousemove', onMouseMove);
                    sticker.onmouseup = null;
                };
            });

            document.body.appendChild(sticker);
        }

            const downloadAnchor = document.createElement('a');
            downloadAnchor.setAttribute("href", dataStr);
            downloadAnchor.setAttribute("download", "kawaii_planner_backup.json");
            document.body.appendChild(downloadAnchor);
            downloadAnchor.click();
            downloadAnchor.remove();
        }

        function importData(event) {
            const reader = new FileReader();
            reader.onload = function(e) {
                appState = JSON.parse(e.target.result);
                saveState();
                location.reload();
            };
            reader.readAsText(event.target.files[0]);
        }
